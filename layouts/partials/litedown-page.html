{{- /* Theme a pre-rendered LiteDown (or similar) document. input: the page; output: safe HTML with the theme bundle, theme bootstrap and a "back to list" link injected into the original document. */ -}} {{- $content := .Content | string -}} {{- $css := partial "css-bundle.html" . -}} {{- $themeInit := partial "theme-init.html" . -}} {{- $backHref := "/" -}} {{- with site.GetPage (printf "/%s" .Section) }}{{ $backHref = .RelPermalink }}{{ end -}} {{- $backLabel := i18n "backToList" -}} {{- $headInject := printf "%s\n\n\n" $themeInit $css.RelPermalink -}} {{- $content = replace $content "" (printf "%s" $headInject) -}} {{- $content = replaceRE `(?s)(]*>)` (printf "%s%s" "$1" $backHref $backLabel) $content -}} {{- return ($content | safeHTML) -}}
