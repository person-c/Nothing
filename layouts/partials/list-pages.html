{{- /* Returns the collection a list page should paginate. Section lists collect every regular page in the section tree: a post may sit directly under content/
/ or inside a subdirectory such as content/blog/chinese/, and both are listed. Nested `_index.md` language lists are navigation, not posts, so they are not paginated themselves. Taxonomy indexes paginate terms. Pages with `listLang` (nested under a section, e.g. content/blog/chinese/_index.md) paginate the matching title-language subset of their parent section (see title-lang-lists.html). */ -}} {{- $listLang := .Params.listLang -}} {{- $base := . -}} {{- if $listLang -}} {{- $parent := .Parent -}} {{- with .Params.listParent -}} {{- with site.GetPage (printf "/%s" (strings.Trim . "/")) }}{{ $parent = . }}{{ end -}} {{- end -}} {{- with $parent }}{{ $base = . }}{{ end -}} {{- end -}} {{- $pages := .Pages -}} {{- if eq .Kind "section" -}} {{- $pages = .RegularPagesRecursive -}} {{- else if and (ne .Kind "taxonomy") (ne .Kind "home") -}} {{- $pages = .RegularPages -}} {{- end -}} {{- if $listLang -}} {{- $langLists := partialCached "title-lang-lists.html" $base $base.RelPermalink -}} {{- $pages = index $langLists $listLang -}} {{- end -}} {{- return $pages -}}
