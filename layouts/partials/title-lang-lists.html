{{- /* Splits a section's regular pages by title language. The whole section tree is classified, so a post nested in a physical language folder (content/blog/chinese/.md) counts exactly like one that sits directly under the section. Usage: {{ partialCached "title-lang-lists.html" $section $section.RelPermalink }} */ -}} {{- $pages := .RegularPages -}} {{- if eq .Kind "section" -}} {{- $pages = .RegularPagesRecursive -}} {{- end -}} {{- $en := slice -}} {{- $zh := slice -}} {{- range $pages -}} {{- $lang := partial "title-lang.html" . -}} {{- if eq $lang "en" -}} {{- $en = $en | append . -}} {{- else -}} {{- $zh = $zh | append . -}} {{- end -}} {{- end -}} {{- return (dict "en" $en "zh" $zh) -}}
