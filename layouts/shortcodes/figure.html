{{- /* figure shortcode, overriding Hugo's built-in. Raw markup inside markdown bypasses the image render hook and therefore got no lazy loading / decoding hints; this shortcode keeps the same features for captioned figures. Usage: {{< figure src="pic.png" alt="Alt text" caption="Caption" >}} {{< figure src="pic.png" alt="Alt" caption="Caption" class="wide" width="1200" >}} */ -}} {{- $src := .Get "src" | default (.Get 0) -}} {{- $alt := .Get "alt" | default (.Get 1) -}} {{- $caption := .Get "caption" | default (.Get 2) -}} {{- $title := .Get "title" -}} {{- $link := .Get "link" -}} {{- $class := .Get "class" -}} {{- $width := .Get "width" -}} {{- $height := .Get "height" -}} {{- $img := .Page.Resources.GetMatch (path.Base $src) -}} {{- if and $img (ne $img.MediaType.SubType "svg") -}} {{- $small := $img.Resize "480x" -}} {{- $large := $img.Resize "960x" -}} {{- with $link }}{{ end -}} {{- with $link }}{{ end -}} {{- with $caption }}{{ end -}} {{- else -}} {{- with $link }}{{ end -}} {{- with $link }}{{ end -}} {{- with $caption }}{{ end -}} {{- end -}}
